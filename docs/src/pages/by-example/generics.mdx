# Generic modules

A generic module like `Set(T : map_key)` is specialized for each element type it is used with. `Set(order_id)` and `Set(address)` are distinct types with their own code.

:::code-group

```lithe [generics.lithe]
(* Generic modules specialize at compile time: no runtime cost, no casts.
   `Set(T : map_key)` accepts any type that can key a map. *)

import "./lib/set.lithe";
import "./lib/set_enumerable.lithe";

type order_id is bytes32;

storage is record {
  admins : Set(address);
  orders : Set(order_id);
  members : SetEnumerable(address);
}

(* Functions can be generic too. *)
pure function first(T)(const a : T; const b : T) : T is
  a

external function add_admin(const who : address) is {
  Set.add(storage.admins, who);
}

external view function is_admin(const who : address) : bool is
  Set.contains(storage.admins, who)

(* `Set(order_id)` is not `Set(bytes32)`: the element type is checked. *)
external function track(const id : bytes32) is {
  Set.add(storage.orders, order_id(id));
}

external function join() is {
  SetEnumerable.add(storage.members, first(Evm.sender(), Evm.tx_origin()));
}

external view function member_count() : uint is
  SetEnumerable.length(storage.members)

external view function member_at(const i : uint) : address
  raises Out_of_bounds
is
  SetEnumerable.at(storage.members, i)

(* A module passes its own storage to a generic module the same way. *)
module Watchlist is {
  storage watchlist at "lithe.watchlist" is record {
    watched : SetEnumerable(address);
  }

  external function watch(const who : address) is {
    SetEnumerable.add(watchlist.watched, who);
  }

  external view function watched_count() : uint is
    SetEnumerable.length(watchlist.watched)
}

use Watchlist exposing {
  function watch(who : address);
  view function watched_count() : uint;
}
```

```lithe [lib/set.lithe]
(* Membership set. One mapping, no enumeration. *)

module Set(T : map_key) is {
  type self is map(T, bool);

  view function contains(storage set : self; const value : T) : bool is
    set[value]

  function add(storage set : self; const value : T) is {
    set[value] := true;
  }

  function remove(storage set : self; const value : T) is {
    set[value] := false;
  }
}
```

```lithe [lib/set_enumerable.lithe]
(* Membership plus enumeration. The list and the index map are both stored. *)

module SetEnumerable(T : map_key) is {
  type self is record {
    values : list(T);
    indexes : map(T, uint);
  }

  type iterator is record {
    index : uint;
  }

  error Out_of_bounds;

  view function contains(storage items : self; const value : T) : bool is
    items.indexes[value] > 0

  view function length(storage items : self) : uint is
    items.values.length

  view function at(storage items : self; const index : uint) : T
    raises Out_of_bounds
  is {
    assert(index < items.values.length, Out_of_bounds);
  } with items.values[index]

  function add(storage items : self; const value : T) is {
    if items.indexes[value] = 0 then {
      List.push(items.values, value);
      items.indexes[value] := items.values.length;
    }
  }

  function remove(storage items : self; const value : T) is {
    const found : uint := items.indexes[value];
    if found = 0 then {
      skip
    } else {
      const moved : T := items.values[items.values.length - 1];
      items.values[found - 1] := moved;
      items.indexes[moved] := found;
      items.indexes[value] := 0;
      List.pop(items.values);
    }
  }
}
```

:::

- `Set.add(storage.admins, who)` works on the storage location it is given. The set module owns no storage of its own.
- `Watchlist` hands `watchlist.watched`, from its own storage, to `SetEnumerable` just as the contract hands over `storage.members`.
- `map_key` limits `T` to types that can key a map. `Set(string)` is rejected.
