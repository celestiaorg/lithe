# Multi-sig wallet

Three owners, two confirmations required. The owners are a fixed-size `list(address, 3)`, so the loop that records them has a length known when the contract is compiled.

```lithe [multisig.lithe]
(* A 2-of-3 wallet for ether payments. Owners submit a payment, confirm it, and
   anyone can execute it once it has enough confirmations. *)

error Not_owner;
error Already_initialized;
error No_such_tx of nat;
error Already_executed of nat;
error Already_confirmed of nat;
error Not_confirmed of nat;
error Not_enough_confirmations of record {
  id : nat;
  confirmations : nat;
}

event Submitted of record {
  indexed id : nat;
  indexed to : address;
  value : wei;
}

event Confirmed of record {
  indexed id : nat;
  indexed owner : address;
}

event Revoked of record {
  indexed id : nat;
  indexed owner : address;
}

event Executed of record {
  indexed id : nat;
}

type payment is record {
  to : address;
  value : wei;
  executed : bool;
  confirmations : nat;
}

storage is record {
  is_owner : map(address, bool);
  threshold : nat;
  count : nat;
  payments : map(nat, payment);
  confirmed : map(nat, map(address, bool));
}

external function init(const owners : list(address, 3))
  raises Already_initialized
is {
  assert(storage.threshold = 0n, Already_initialized);
  for owner : address in owners do {
    storage.is_owner[owner] := true;
  }
  storage.threshold := 2n;
}

external payable function deposit() is {
  skip
}

view function only_owner() is {
  assert(storage.is_owner[Evm.sender()], Not_owner);
}

view function pending(const id : nat) is {
  assert(id < storage.count, No_such_tx(id));
  assert(not storage.payments[id].executed, Already_executed(id));
}

external function submit(const to : address; const value : wei) : nat
  raises Not_owner
is {
  only_owner();
  const id : nat := storage.count;
  storage.payments[id] := { to = to; value = value; executed = false; confirmations = 0n; };
  storage.count := id + 1n;
  emit Submitted { id = id; to = to; value = value; };
} with id

external function confirm(const id : nat)
  raises Not_owner | No_such_tx | Already_executed | Already_confirmed
is {
  only_owner();
  pending(id);
  assert(not storage.confirmed[id][Evm.sender()], Already_confirmed(id));
  storage.confirmed[id][Evm.sender()] := true;
  storage.payments[id].confirmations := storage.payments[id].confirmations + 1n;
  emit Confirmed { id = id; owner = Evm.sender(); };
}

external function revoke(const id : nat)
  raises Not_owner | No_such_tx | Already_executed | Not_confirmed
is {
  only_owner();
  pending(id);
  assert(storage.confirmed[id][Evm.sender()], Not_confirmed(id));
  storage.confirmed[id][Evm.sender()] := false;
  storage.payments[id].confirmations := storage.payments[id].confirmations - 1n;
  emit Revoked { id = id; owner = Evm.sender(); };
}

external nonreentrant function execute(const id : nat)
  raises No_such_tx | Already_executed | Not_enough_confirmations
is {
  pending(id);
  const p : payment := storage.payments[id];
  assert(
    p.confirmations >= storage.threshold,
    Not_enough_confirmations { id = id; confirmations = p.confirmations; }
  );
  storage.payments[id].executed := true;
  emit Executed { id = id; };
  defer Evm.transfer(p.to, p.value);
}

external view function payment_of(const id : nat) : payment is
  storage.payments[id]
```

- `execute` marks the payment executed before the deferred transfer runs, and `nonreentrant` stops the recipient from calling `execute` again.
- This wallet sends ether only. Lithe has no arbitrary-calldata calls, so a general-purpose multisig would dispatch typed calls through interfaces.
