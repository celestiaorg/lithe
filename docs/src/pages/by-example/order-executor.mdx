# Order executor

A larger continuation example. `execute_and_deposit` swaps, and its continuation deposits the swap's output, and that continuation's own continuation records the shares. `unwind` destructures a tuple result.

```lithe [order-executor.lithe]
(* An order executor that swaps on an exchange and deposits the proceeds, using
   each call's result in a continuation. The order is marked pending before the
   exchange runs, so the exchange, and anything it calls, sees a consistent state. *)

type swap_result is record {
  amount_out : nat;
  fee : nat;
}

interface Exchange is {
  function swap(amount_in : nat; min_out : nat) : swap_result;
  function deposit(amount : nat) : nat;
  function split(shares : nat) : (nat, nat);
}

type order is record {
  status : uint8;
  amount_in : nat;
  amount_out : nat;
  shares : nat;
}

error AlreadyExecuted of nat;
error NotComplete of nat;
error TooLittle of record {
  amount_out : nat;
  floor : nat;
}

event Executed of record {
  indexed id : nat;
  amount_out : nat;
  fee : nat;
}

event Deposited of record {
  indexed id : nat;
  shares : nat;
}

storage is record {
  exchange : Exchange;
  floor : nat;
  executed : nat;
  total_out : nat;
  left : nat;
  right : nat;
  orders : map(nat, order);
}

external function init(const exchange : Exchange; const floor : nat) is {
  storage.exchange := exchange;
  storage.floor := floor;
}

external nonreentrant function execute(const id : nat; const amount_in : nat; const min_out : nat)
  raises AlreadyExecuted | TooLittle
is {
  assert(storage.orders[id].status = uint8(0), AlreadyExecuted(id));
  storage.orders[id].status := uint8(1);
  storage.orders[id].amount_in := amount_in;
  defer storage.exchange.swap(amount_in, min_out)
    then (result : swap_result) {
      assert(
        result.amount_out >= storage.floor,
        TooLittle { amount_out = result.amount_out; floor = storage.floor; }
      );
      storage.orders[id].status := uint8(2);
      storage.orders[id].amount_out := result.amount_out;
      storage.executed := storage.executed + 1n;
      storage.total_out := storage.total_out + result.amount_out;
      emit Executed { id = id; amount_out = result.amount_out; fee = result.fee; };
    };
}

external nonreentrant function execute_and_deposit(const id : nat; const amount_in : nat)
  raises AlreadyExecuted | TooLittle
is {
  assert(storage.orders[id].status = uint8(0), AlreadyExecuted(id));
  storage.orders[id].status := uint8(1);
  storage.orders[id].amount_in := amount_in;
  defer storage.exchange.swap(amount_in, 0n)
    then (result : swap_result) {
      storage.orders[id].status := uint8(2);
      storage.orders[id].amount_out := result.amount_out;
      defer storage.exchange.deposit(result.amount_out)
        then (shares : nat) {
          storage.orders[id].shares := shares;
          emit Deposited { id = id; shares = shares; };
        };
    };
}

external nonreentrant function unwind(const id : nat)
  raises NotComplete
is {
  assert(storage.orders[id].status = uint8(2), NotComplete(id));
  defer storage.exchange.split(storage.orders[id].shares)
    then ((a, b) : (nat, nat)) {
      storage.orders[id].status := uint8(3);
      storage.left := storage.left + a;
      storage.right := storage.right + b;
    };
}

external view function order_of(const id : nat) : order is storage.orders[id]
```

- Each `then` is a separate phase. The exchange sees each order as pending (`status = 1`) while it runs.
- `storage.floor` is re-read in the continuation, so if the exchange changed it, the check uses the new value.
